# Analytics SDK & Dashboard Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build a working end-to-end analytics system (browser SDK, server SDK, ingestion API + SQLite, dashboard) and a restaurant demo app that exercises it.

**Architecture:** npm-workspaces monorepo. `packages/shared` holds the event contract (types + Zod schemas). `packages/browser-sdk` and `packages/server-sdk` are consumed as TypeScript source by two Next.js apps: `apps/analytics` (ingestion API, stats API, dashboard, SQLite via better-sqlite3, port 3000) and `apps/restaurant` (demo site, port 3001). Business logic lives in plain functions that take a `db` argument so it is unit-testable; Next.js route files are thin wrappers.

**Tech Stack:** Node 20+, TypeScript 5, Next.js 15 (App Router), React 19, better-sqlite3 11, Zod 3, Recharts 2, Vitest 3, jsdom, tsx, concurrently.

**Spec:** `docs/superpowers/specs/2026-10-16-analytics-sdk-dashboard-design.md`

## Global Constraints

- Node ≥ 20; all code TypeScript, `strict: true`.
- Batch: 1–50 events; body ≤ 64 KB; properties ≤ 8 KB; event name `^[a-zA-Z0-9_.:-]{1,100}$`.
- `event_id` is a UUID generated by the SDK; storage is idempotent on `(project_id, event_id)`.
- Clock skew: client timestamp > 24 h in the future or > 7 days in the past → use server `received_at`.
- Session timeout 30 min; browser queue cap 500; flush every 5000 ms or at 20 events; backoff 1s,2s,4s… capped 30 s (+≤20% jitter).
- Server SDK: timeout 2000 ms, 2 retries, never throws.
- Rate limit: 120 requests/min per (key, IP), in-memory.
- IP addresses are never stored.
- Reporting uses UTC days; date ranges inclusive; max 366 days.
- Demo defaults (work without `.env`): project `demo`, public key `pk_demo_123`, secret key `sk_demo_456`, allowed origin `http://localhost:3001`, dashboard password `admin`.
- SDKs never throw into host code.
- Use relative imports (no `@/` alias) so Vitest needs no path config.

## Review Focus

1. **Analytics server is down while a customer orders** → the order is still created and the response is not delayed (tracking is deferred). Test: Task 9 `create-order.test.ts` "order succeeds when analytics fails".
2. **The same events arrive twice** (beacon on page hide + later normal flush) → stored once. Test: Task 3 "duplicate event_id is stored once".
3. **localStorage throws** (Safari private mode, quota) → SDK still tracks and sends. Test: Task 7 "works when localStorage throws".
4. **Disallowed origin / bad key** → SDK receives a readable 4xx (CORS echoed) and drops the batch instead of retrying forever. Tests: Task 4 "403 keeps CORS header", Task 7 "drops batch on 4xx".
5. **Bad custom date range in the dashboard URL** → clear error, fall back to default range instead of crashing. Test: Task 5 `parseRange` cases; dashboard page catches the error.

```ts
import { z } from 'zod';
import { EVENT_NAME_PATTERN, MAX_BATCH_EVENTS, MAX_PROPERTIES_BYTES } from './limits';

const byteLength = (s: string) => new TextEncoder().encode(s).length;
const optionalText = (max: number) => z.string().max(max).nullish();

export const eventSchema = z
  .object({
    event_id: z.string().uuid(),
    type: z.enum(['page_view', 'track', 'identify']),
    name: z.string().regex(EVENT_NAME_PATTERN, 'must be 1-100 chars of letters, digits, _ . : -'),
    anonymous_id: z.string().min(1).max(100).nullish(),
    user_id: z.string().min(1).max(200).nullish(),
    session_id: optionalText(100),
    timestamp: z.string().datetime({ offset: true }),
    url: optionalText(2048),
    title: optionalText(500),
    referrer: optionalText(2048),
    properties: z.record(z.unknown()).default({}),
  })
  .superRefine((e, ctx) => {
    if (!e.anonymous_id && !e.user_id) {
      ctx.addIssue({ code: z.ZodIssueCode.custom, path: ['anonymous_id'], message: 'anonymous_id or user_id is required' });
    }
    if (byteLength(JSON.stringify(e.properties)) > MAX_PROPERTIES_BYTES) {
      ctx.addIssue({ code: z.ZodIssueCode.custom, path: ['properties'], message: `must be at most ${MAX_PROPERTIES_BYTES} bytes` });
    }
  });

export type ValidEvent = z.output<typeof eventSchema>;

export const batchSchema = z.object({
  publicKey: z.string().optional(),
  events: z.array(z.unknown()).min(1).max(MAX_BATCH_EVENTS),
});

export function formatZodError(err: z.ZodError): string {
  return err.issues.map((i) => `${i.path.join('.') || 'body'}: ${i.message}`).join('; ');
}
```

```ts
import Database from 'better-sqlite3';
import fs from 'node:fs';
import path from 'node:path';
import { config } from './config';
import { upsertProject } from './projects';

export type DB = Database.Database;

const SCHEMA = `
CREATE TABLE IF NOT EXISTS projects (
  id              TEXT PRIMARY KEY,
  name            TEXT NOT NULL,
  public_key      TEXT NOT NULL UNIQUE,
  secret_key_hash TEXT NOT NULL UNIQUE,
  allowed_origins TEXT NOT NULL,
  created_at      TEXT NOT NULL
);

CREATE TABLE IF NOT EXISTS events (
  id           INTEGER PRIMARY KEY AUTOINCREMENT,
  event_id     TEXT NOT NULL,
  project_id   TEXT NOT NULL REFERENCES projects(id),
  type         TEXT NOT NULL,
  name         TEXT NOT NULL,
  source       TEXT NOT NULL,
  anonymous_id TEXT,
  user_id      TEXT,
  session_id   TEXT,
  url          TEXT,
  title        TEXT,
  referrer     TEXT,
  user_agent   TEXT,
  properties   TEXT NOT NULL DEFAULT '{}',
  occurred_at  TEXT NOT NULL,
  received_at  TEXT NOT NULL,
  UNIQUE (project_id, event_id)
);

CREATE INDEX IF NOT EXISTS idx_events_project_time ON events (project_id, occurred_at);
CREATE INDEX IF NOT EXISTS idx_events_project_name_time ON events (project_id, name, occurred_at);
`;

export function openDb(file: string): DB {
  if (file !== ':memory:') fs.mkdirSync(path.dirname(path.resolve(file)), { recursive: true });
  const db = new Database(file);
  db.pragma('journal_mode = WAL');
  db.pragma('foreign_keys = ON');
  db.pragma('busy_timeout = 5000');
  db.exec(SCHEMA);
  return db;
}

// Cached on globalThis so Next.js hot reloads reuse one connection.
const globalForDb = globalThis as unknown as { __analyticsDb?: DB };

export function getDb(): DB {
  if (!globalForDb.__analyticsDb) {
    const db = openDb(config.dbPath);
    upsertProject(db, config.demoProject);
    globalForDb.__analyticsDb = db;
  }
  return globalForDb.__analyticsDb;
}

export function setDb(db?: DB): void {
  globalForDb.__analyticsDb = db;
}
```
`apps/analytics/src/lib/projects.ts`
```ts
import { createHash } from 'node:crypto';
import type { DB } from './db';

export interface Project {
  id: string;
  name: string;
  publicKey: string;
  allowedOrigins: string[];
}

export interface ProjectInput extends Project {
  secretKey: string;
}

interface ProjectRow {
  id: string;
  name: string;
  public_key: string;
  allowed_origins: string;
}

// Secret keys are long random strings, so a plain SHA-256 is enough (no need for bcrypt).
export const hashSecret = (secret: string) => createHash('sha256').update(secret).digest('hex');

const toProject = (row: ProjectRow): Project => ({
  id: row.id,
  name: row.name,
  publicKey: row.public_key,
  allowedOrigins: JSON.parse(row.allowed_origins) as string[],
});

export function upsertProject(db: DB, p: ProjectInput): void {
  db.prepare(
    `INSERT INTO projects (id, name, public_key, secret_key_hash, allowed_origins, created_at)
     VALUES (@id, @name, @publicKey, @secretHash, @origins, @now)
     ON CONFLICT(id) DO UPDATE SET
       name = excluded.name,
       public_key = excluded.public_key,
       secret_key_hash = excluded.secret_key_hash,
       allowed_origins = excluded.allowed_origins`,
  ).run({
    id: p.id,
    name: p.name,
    publicKey: p.publicKey,
    secretHash: hashSecret(p.secretKey),
    origins: JSON.stringify(p.allowedOrigins),
    now: new Date().toISOString(),
  });
}

const SELECT = 'SELECT id, name, public_key, allowed_origins FROM projects';

export function findProjectByPublicKey(db: DB, publicKey: string): Project | null {
  const row = db.prepare(`${SELECT} WHERE public_key = ?`).get(publicKey) as ProjectRow | undefined;
  return row ? toProject(row) : null;
}

export function findProjectBySecretKey(db: DB, secretKey: string): Project | null {
  const row = db.prepare(`${SELECT} WHERE secret_key_hash = ?`).get(hashSecret(secretKey)) as ProjectRow | undefined;
  return row ? toProject(row) : null;
}
```

```ts
import { eventSchema, formatZodError, type EventSource, type IngestResponse } from '@acme/shared';
import type { DB } from './db';

const MAX_FUTURE_MS = 24 * 60 * 60 * 1000;
const MAX_PAST_MS = 7 * 24 * 60 * 60 * 1000;

/** Trust the client timestamp unless its clock is clearly wrong. Always returns UTC ISO. */
export function normaliseOccurredAt(clientTs: string, receivedAt: Date): string {
  const t = Date.parse(clientTs);
  const now = receivedAt.getTime();
  if (Number.isNaN(t) || t > now + MAX_FUTURE_MS || t < now - MAX_PAST_MS) return receivedAt.toISOString();
  return new Date(t).toISOString();
}

export interface IngestInput {
  projectId: string;
  source: EventSource;
  events: unknown[];
  userAgent?: string | null;
  now?: Date;
}

export function ingestEvents(db: DB, input: IngestInput): IngestResponse {
  const receivedAt = input.now ?? new Date();
  const result: IngestResponse = { accepted: 0, duplicates: 0, rejected: [] };
  const rows: Record<string, unknown>[] = [];

  input.events.forEach((raw, index) => {
    const parsed = eventSchema.safeParse(raw);
    if (!parsed.success) {
      result.rejected.push({ index, error: formatZodError(parsed.error) });
      return;
    }
    const e = parsed.data;
    if (input.source === 'browser' && !e.anonymous_id) {
      result.rejected.push({ index, error: 'anonymous_id: required for browser events' });
      return;
    }
    rows.push({
      event_id: e.event_id,
      project_id: input.projectId,
      type: e.type,
      name: e.name,
      source: input.source,
      anonymous_id: e.anonymous_id ?? null,
      user_id: e.user_id ?? null,
      session_id: e.session_id ?? null,
      url: e.url ?? null,
      title: e.title ?? null,
      referrer: e.referrer ?? null,
      user_agent: input.userAgent ? input.userAgent.slice(0, 512) : null,
      properties: JSON.stringify(e.properties),
      occurred_at: normaliseOccurredAt(e.timestamp, receivedAt),
      received_at: receivedAt.toISOString(),
    });
  });

  // INSERT OR IGNORE + UNIQUE(project_id, event_id) makes retries safe (idempotent).
  const insert = db.prepare(
    `INSERT OR IGNORE INTO events
       (event_id, project_id, type, name, source, anonymous_id, user_id, session_id,
        url, title, referrer, user_agent, properties, occurred_at, received_at)
     VALUES
       (@event_id, @project_id, @type, @name, @source, @anonymous_id, @user_id, @session_id,
        @url, @title, @referrer, @user_agent, @properties, @occurred_at, @received_at)`,
  );
  db.transaction(() => {
    for (const row of rows) {
      if (insert.run(row).changes === 1) result.accepted++;
      else result.duplicates++;
    }
  })();

  return result;
}
```

```ts
export class HttpError extends Error {
  constructor(public status: number, message: string) {
    super(message);
  }
}

/** Reads the body as text (works for application/json and text/plain from sendBeacon) and parses JSON. */
export async function readJsonBody(req: Request, maxBytes: number): Promise<unknown> {
  const declared = Number(req.headers.get('content-length') ?? 0);
  if (declared > maxBytes) throw new HttpError(413, 'Request body too large');
  const text = await req.text();
  if (new TextEncoder().encode(text).length > maxBytes) throw new HttpError(413, 'Request body too large');
  try {
    return JSON.parse(text);
  } catch {
    throw new HttpError(400, 'Body must be valid JSON');
  }
}

export function jsonResponse(data: unknown, status = 200, headers: Record<string, string> = {}): Response {
  return Response.json(data, { status, headers });
}

export function errorResponse(err: unknown, headers: Record<string, string> = {}): Response {
  if (err instanceof HttpError) return jsonResponse({ error: err.message }, err.status, headers);
  console.error('[analytics] unexpected error', err);
  return jsonResponse({ error: 'Internal server error' }, 500, headers);
}

/** Used only for rate limiting, never stored. */
export function clientIp(req: Request): string {
  return req.headers.get('x-forwarded-for')?.split(',')[0].trim() || req.headers.get('x-real-ip') || 'unknown';
}
```

```ts
export interface RateLimiter {
  check(key: string, now?: number): { allowed: boolean; retryAfterSec: number };
}

/** Fixed-window counter kept in memory. Per process only — use Redis when running several servers. */
export function createRateLimiter({ limit, windowMs }: { limit: number; windowMs: number }): RateLimiter {
  const windows = new Map<string, { start: number; count: number }>();

  function prune(now: number) {
    for (const [key, w] of windows) if (now - w.start >= windowMs) windows.delete(key);
  }

  return {
    check(key, now = Date.now()) {
      let w = windows.get(key);
      if (!w || now - w.start >= windowMs) {
        w = { start: now, count: 0 };
        windows.set(key, w);
        if (windows.size > 10_000) prune(now);
      }
      w.count++;
      if (w.count > limit) {
        return { allowed: false, retryAfterSec: Math.ceil((w.start + windowMs - now) / 1000) };
      }
      return { allowed: true, retryAfterSec: 0 };
    },
  };
}
```

```ts
import { batchSchema, formatZodError, MAX_BODY_BYTES } from '@acme/shared';
import type { DB } from './db';
import { ingestEvents } from './ingest';
import { findProjectByPublicKey, findProjectBySecretKey, type Project } from './projects';
import { clientIp, errorResponse, HttpError, jsonResponse, readJsonBody } from './http';
import { createRateLimiter, type RateLimiter } from './rate-limit';

export const ingestLimiter = createRateLimiter({ limit: 120, windowMs: 60_000 });

/**
 * The collect endpoint only accepts writes and returns no private data, so we echo the
 * caller's Origin. That lets the SDK read 401/403 responses and drop the batch instead of
 * retrying a CORS failure forever. The real origin check happens below.
 */
function corsHeaders(origin: string | null): Record<string, string> {
  if (!origin) return {};
  return {
    'Access-Control-Allow-Origin': origin,
    'Access-Control-Allow-Methods': 'POST, OPTIONS',
    'Access-Control-Allow-Headers': 'Content-Type, X-Analytics-Key',
    'Access-Control-Max-Age': '86400',
    Vary: 'Origin',
  };
}

async function parseBatch(req: Request) {
  const body = await readJsonBody(req, MAX_BODY_BYTES);
  const envelope = batchSchema.safeParse(body);
  if (!envelope.success) throw new HttpError(400, formatZodError(envelope.error));
  return envelope.data;
}

function enforceRateLimit(limiter: RateLimiter, key: string, headers: Record<string, string>): Response | null {
  const rl = limiter.check(key);
  if (rl.allowed) return null;
  return jsonResponse({ error: 'Rate limit exceeded' }, 429, { ...headers, 'Retry-After': String(rl.retryAfterSec) });
}

export function handleCollectPreflight(req: Request): Response {
  return new Response(null, { status: 204, headers: corsHeaders(req.headers.get('origin')) });
}

export async function handleCollect(req: Request, db: DB, limiter: RateLimiter = ingestLimiter): Promise<Response> {
  const origin = req.headers.get('origin');
  const cors = corsHeaders(origin);
  try {
    const batch = await parseBatch(req);
    const key = req.headers.get('x-analytics-key') ?? batch.publicKey;
    const project: Project | null = key ? findProjectByPublicKey(db, key) : null;
    if (!project) throw new HttpError(401, 'Invalid public key');
    if (!origin || !project.allowedOrigins.includes(origin)) throw new HttpError(403, 'Origin not allowed');

    const limited = enforceRateLimit(limiter, `${project.publicKey}|${clientIp(req)}`, cors);
    if (limited) return limited;

    const result = ingestEvents(db, {
      projectId: project.id,
      source: 'browser',
      events: batch.events,
      userAgent: req.headers.get('user-agent'),
    });
    return jsonResponse(result, 202, cors);
  } catch (err) {
    return errorResponse(err, cors);
  }
}

export async function handleServerEvents(req: Request, db: DB, limiter: RateLimiter = ingestLimiter): Promise<Response> {
  try {
    const auth = req.headers.get('authorization') ?? '';
    const secret = auth.startsWith('Bearer ') ? auth.slice(7).trim() : '';
    const project = secret ? findProjectBySecretKey(db, secret) : null;
    if (!project) throw new HttpError(401, 'Invalid secret key');

    const limited = enforceRateLimit(limiter, `server:${project.id}|${clientIp(req)}`, {});
    if (limited) return limited;

    const batch = await parseBatch(req);
    const result = ingestEvents(db, { projectId: project.id, source: 'server', events: batch.events });
    return jsonResponse(result, 202);
  } catch (err) {
    return errorResponse(err);
  }
}
```

```ts
import type { DB } from './db';

export interface DateRange {
  from: string; // YYYY-MM-DD, inclusive (UTC)
  to: string;   // YYYY-MM-DD, inclusive (UTC)
}

export class InvalidRangeError extends Error {}

const DAY_MS = 86_400_000;
const MAX_DAYS = 366;
const DATE_RE = /^\d{4}-\d{2}-\d{2}$/;

const toDateStr = (d: Date) => d.toISOString().slice(0, 10);
const dayStart = (date: string) => Date.parse(`${date}T00:00:00.000Z`);

function validDate(value: string): boolean {
  // Round-trip check rejects impossible dates such as 2026-02-31.
  return DATE_RE.test(value) && !Number.isNaN(dayStart(value)) && toDateStr(new Date(dayStart(value))) === value;
}

export function parseRange(params: URLSearchParams, today: Date = new Date()): DateRange {
  const to = params.get('to') || toDateStr(today);
  if (!validDate(to)) throw new InvalidRangeError(`Invalid "to" date: ${to}`);
  const from = params.get('from') || toDateStr(new Date(dayStart(to) - 6 * DAY_MS));
  if (!validDate(from)) throw new InvalidRangeError(`Invalid "from" date: ${from}`);
  if (from > to) throw new InvalidRangeError('"from" must be on or before "to"');
  if ((dayStart(to) - dayStart(from)) / DAY_MS + 1 > MAX_DAYS) throw new InvalidRangeError(`Range cannot exceed ${MAX_DAYS} days`);
  return { from, to };
}

export function presetRanges(today: Date = new Date()) {
  const to = toDateStr(today);
  const back = (days: number) => toDateStr(new Date(dayStart(to) - days * DAY_MS));
  return [
    { label: 'Today', from: to, to },
    { label: 'Last 7 days', from: back(6), to },
    { label: 'Last 30 days', from: back(29), to },
    { label: 'Last 90 days', from: back(89), to },
  ];
}

export function listDays(range: DateRange): string[] {
  const days: string[] = [];
  for (let t = dayStart(range.from); t <= dayStart(range.to); t += DAY_MS) days.push(toDateStr(new Date(t)));
  return days;
}

/** Half-open interval [start, end) on occurred_at; ISO strings compare correctly as text. */
function bounds(projectId: string, range: DateRange) {
  return {
    projectId,
    start: new Date(dayStart(range.from)).toISOString(),
    end: new Date(dayStart(range.to) + DAY_MS).toISOString(),
  };
}

const WHERE = 'project_id = @projectId AND occurred_at >= @start AND occurred_at < @end';

export interface Overview {
  pageViews: number;
  uniqueVisitors: number;
  sessions: number;
  events: number;
  orders: number;
  revenue: number;
}

export function getOverview(db: DB, projectId: string, range: DateRange): Overview {
  const row = db
    .prepare(
      `SELECT
         COALESCE(SUM(type = 'page_view'), 0) AS pageViews,
         COUNT(DISTINCT CASE WHEN source = 'browser' THEN anonymous_id END) AS uniqueVisitors,
         COUNT(DISTINCT session_id) AS sessions,
         COALESCE(SUM(type = 'track'), 0) AS events,
         COALESCE(SUM(name = 'order_created'), 0) AS orders,
         COALESCE(SUM(CASE WHEN name = 'order_created' THEN json_extract(properties, '$.total') END), 0) AS revenue
       FROM events WHERE ${WHERE}`,
    )
    .get(bounds(projectId, range)) as Overview;
  return { ...row, revenue: Math.round(row.revenue * 100) / 100 };
}

export interface TimeseriesPoint {
  date: string;
  pageViews: number;
  uniqueVisitors: number;
}

export function getTimeseries(db: DB, projectId: string, range: DateRange): TimeseriesPoint[] {
  const rows = db
    .prepare(
      `SELECT substr(occurred_at, 1, 10) AS date,
              SUM(type = 'page_view') AS pageViews,
              COUNT(DISTINCT CASE WHEN source = 'browser' THEN anonymous_id END) AS uniqueVisitors
       FROM events WHERE ${WHERE}
       GROUP BY date`,
    )
    .all(bounds(projectId, range)) as TimeseriesPoint[];
  const byDate = new Map(rows.map((r) => [r.date, r]));
  return listDays(range).map((date) => byDate.get(date) ?? { date, pageViews: 0, uniqueVisitors: 0 });
}

export interface TopPage {
  url: string;
  views: number;
  uniqueVisitors: number;
}

export function getTopPages(db: DB, projectId: string, range: DateRange, limit = 10): TopPage[] {
  return db
    .prepare(
      `SELECT url, COUNT(*) AS views, COUNT(DISTINCT anonymous_id) AS uniqueVisitors
       FROM events
       WHERE ${WHERE} AND type = 'page_view' AND url IS NOT NULL
       GROUP BY url
       ORDER BY views DESC, url ASC
       LIMIT @limit`,
    )
    .all({ ...bounds(projectId, range), limit }) as TopPage[];
}

export interface EventTrends {
  names: string[];
  series: Record<string, string | number>[];
}

export function getEventTrends(db: DB, projectId: string, range: DateRange): EventTrends {
  const rows = db
    .prepare(
      `SELECT substr(occurred_at, 1, 10) AS date, name, COUNT(*) AS count
       FROM events WHERE ${WHERE} AND type = 'track'
       GROUP BY date, name`,
    )
    .all(bounds(projectId, range)) as { date: string; name: string; count: number }[];

  const totals = new Map<string, number>();
  for (const r of rows) totals.set(r.name, (totals.get(r.name) ?? 0) + r.count);
  const names = [...totals.keys()].sort((a, b) => totals.get(b)! - totals.get(a)! || a.localeCompare(b));

  const series = listDays(range).map((date) => {
    const point: Record<string, string | number> = { date };
    for (const n of names) point[n] = 0;
    return point;
  });
  const byDate = new Map(series.map((p) => [p.date as string, p]));
  for (const r of rows) byDate.get(r.date)![r.name] = r.count;
  return { names, series };
}
```

```ts
import { createHash, createHmac, timingSafeEqual } from 'node:crypto';

export const SESSION_COOKIE = 'dash_session';
const TTL_MS = 8 * 60 * 60 * 1000;

const sign = (payload: string, secret: string) => createHmac('sha256', secret).update(payload).digest('hex');

function safeEqual(a: string, b: string): boolean {
  const ab = Buffer.from(a);
  const bb = Buffer.from(b);
  return ab.length === bb.length && timingSafeEqual(ab, bb);
}

/** Token = "<expiry ms>.<hmac(expiry)>". Stateless: no session table needed. */
export function createSessionToken(secret: string, now = Date.now()): string {
  const expiry = String(now + TTL_MS);
  return `${expiry}.${sign(expiry, secret)}`;
}

export function verifySessionToken(token: string | undefined, secret: string, now = Date.now()): boolean {
  if (!token) return false;
  const [expiry, signature] = token.split('.');
  if (!expiry || !signature) return false;
  if (!safeEqual(signature, sign(expiry, secret))) return false;
  return Number(expiry) > now;
}

/** Hash both sides first so the comparison is constant-time regardless of length. */
export function checkPassword(input: string, expected: string): boolean {
  const h = (s: string) => createHash('sha256').update(s).digest('hex');
  return input.length > 0 && safeEqual(h(input), h(expected));
}

export function readCookie(req: Request, name: string): string | undefined {
  const header = req.headers.get('cookie') ?? '';
  for (const part of header.split(';')) {
    const [key, ...rest] = part.trim().split('=');
    if (key === name) return decodeURIComponent(rest.join('='));
  }
  return undefined;
}

export function sessionCookieHeader(token: string): string {
  return `${SESSION_COOKIE}=${token}; Path=/; HttpOnly; SameSite=Lax; Max-Age=${TTL_MS / 1000}`;
}

export function clearSessionCookieHeader(): string {
  return `${SESSION_COOKIE}=; Path=/; HttpOnly; SameSite=Lax; Max-Age=0`;
}
```

```ts
import type { DB } from './db';
import { readCookie, SESSION_COOKIE, verifySessionToken } from './auth';
import { jsonResponse } from './http';
import { getEventTrends, getOverview, getTimeseries, getTopPages, InvalidRangeError, parseRange } from './stats';

export const STAT_KINDS = ['overview', 'timeseries', 'top-pages', 'events'] as const;
export type StatKind = (typeof STAT_KINDS)[number];

export function handleStats(req: Request, db: DB, projectId: string, kind: StatKind, secret: string): Response {
  if (!verifySessionToken(readCookie(req, SESSION_COOKIE), secret)) {
    return jsonResponse({ error: 'Unauthorized' }, 401);
  }
  try {
    const params = new URL(req.url).searchParams;
    const range = parseRange(params);
    switch (kind) {
      case 'overview':
        return jsonResponse(getOverview(db, projectId, range));
      case 'timeseries':
        return jsonResponse(getTimeseries(db, projectId, range));
      case 'top-pages': {
        const limit = Math.min(100, Math.max(1, Number(params.get('limit')) || 10));
        return jsonResponse(getTopPages(db, projectId, range, limit));
      }
      case 'events':
        return jsonResponse(getEventTrends(db, projectId, range));
    }
  } catch (err) {
    if (err instanceof InvalidRangeError) return jsonResponse({ error: err.message }, 400);
    throw err;
  }
}
```

```ts
/**
 * localStorage with an in-memory fallback (Safari private mode, disabled storage, quota errors).
 * Never throws.
 */
const memory = new Map<string, string>();

export const storage = {
  get(key: string): string | null {
    try {
      return window.localStorage.getItem(key);
    } catch {
      return memory.get(key) ?? null;
    }
  },
  set(key: string, value: string): void {
    try {
      window.localStorage.setItem(key, value);
    } catch {
      memory.set(key, value);
    }
  },
  remove(key: string): void {
    memory.delete(key);
    try {
      window.localStorage.removeItem(key);
    } catch {
      /* ignore */
    }
  },
};

export function getCookie(name: string): string | null {
  try {
    const match = document.cookie.match(new RegExp(`(?:^|; )${name}=([^;]*)`));
    return match ? decodeURIComponent(match[1]) : null;
  } catch {
    return null;
  }
}

export function setCookie(name: string, value: string, maxAgeSeconds: number): void {
  try {
    document.cookie = `${name}=${encodeURIComponent(value)}; Max-Age=${maxAgeSeconds}; Path=/; SameSite=Lax`;
  } catch {
    /* ignore */
  }
}
```

```ts
import { getCookie, setCookie, storage } from './storage';

const ANON_KEY = 'acme_anon_id';
const SESSION_KEY = 'acme_session';
const USER_KEY = 'acme_user_id';
const OPT_OUT_KEY = 'acme_opt_out';
const ONE_YEAR_S = 365 * 24 * 60 * 60;
export const SESSION_TIMEOUT_MS = 30 * 60 * 1000;

/** RFC 4122 v4 id. crypto.randomUUID is missing on plain-http (non-secure) pages, so fall back. */
export function uuid(): string {
  if (typeof crypto !== 'undefined' && typeof crypto.randomUUID === 'function') return crypto.randomUUID();
  return 'xxxxxxxx-xxxx-4xxx-yxxx-xxxxxxxxxxxx'.replace(/[xy]/g, (c) => {
    const r = (Math.random() * 16) | 0;
    return (c === 'x' ? r : (r & 0x3) | 0x8).toString(16);
  });
}

export class Identity {
  getAnonymousId(): string {
    let id = storage.get(ANON_KEY);
    if (!id) {
      id = getCookie(ANON_KEY) ?? `anon_${uuid()}`;
      storage.set(ANON_KEY, id);
      setCookie(ANON_KEY, id, ONE_YEAR_S);
    }
    return id;
  }

  /** Returns the current session id, starting a new one after 30 minutes of inactivity. */
  getSessionId(now = Date.now()): string {
    let session: { id: string; lastActivity: number } | null = null;
    try {
      session = JSON.parse(storage.get(SESSION_KEY) ?? 'null');
    } catch {
      session = null;
    }
    if (!session || typeof session.id !== 'string' || now - session.lastActivity > SESSION_TIMEOUT_MS) {
      session = { id: `sess_${uuid()}`, lastActivity: now };
    } else {
      session.lastActivity = now;
    }
    storage.set(SESSION_KEY, JSON.stringify(session));
    return session.id;
  }

  getUserId(): string | null {
    return storage.get(USER_KEY);
  }

  setUserId(userId: string): void {
    storage.set(USER_KEY, userId);
  }

  /** Logout: forget the user and start a fresh anonymous identity so the next person is not linked. */
  reset(): void {
    storage.remove(USER_KEY);
    storage.remove(ANON_KEY);
    storage.remove(SESSION_KEY);
    setCookie(ANON_KEY, '', 0);
  }

  isOptedOut(): boolean {
    return storage.get(OPT_OUT_KEY) === '1';
  }

  setOptOut(value: boolean): void {
    if (value) storage.set(OPT_OUT_KEY, '1');
    else storage.remove(OPT_OUT_KEY);
  }
}
```

```ts
import type { AnalyticsEvent } from '@acme/shared';
import { storage } from './storage';

/** FIFO queue mirrored to storage so events survive reloads and offline periods. */
export class PersistentQueue {
  private items: AnalyticsEvent[];

  constructor(private readonly key: string, private readonly max = 500) {
    this.items = this.load();
  }

  private load(): AnalyticsEvent[] {
    try {
      const parsed = JSON.parse(storage.get(this.key) ?? '[]');
      return Array.isArray(parsed) ? parsed.slice(-this.max) : [];
    } catch {
      return [];
    }
  }

  private save(): void {
    storage.set(this.key, JSON.stringify(this.items));
  }

  push(event: AnalyticsEvent): void {
    this.items.push(event);
    if (this.items.length > this.max) this.items.splice(0, this.items.length - this.max); // drop oldest
    this.save();
  }

  peek(n: number): AnalyticsEvent[] {
    return this.items.slice(0, n);
  }

  remove(eventIds: string[]): void {
    const ids = new Set(eventIds);
    this.items = this.items.filter((e) => !ids.has(e.event_id));
    this.save();
  }

  clear(): void {
    this.items = [];
    this.save();
  }

  get size(): number {
    return this.items.length;
  }
}
```

```ts
import type { AnalyticsEvent } from '@acme/shared';

export type SendResult = 'ok' | 'retry' | 'drop';

/**
 * text/plain keeps this a CORS "simple request" (no preflight) and matches sendBeacon.
 * 'retry' for network errors, 429 and 5xx; 'drop' for other 4xx because resending cannot succeed.
 */
export async function sendBatch(fetchFn: typeof fetch, url: string, publicKey: string, events: AnalyticsEvent[]): Promise<SendResult> {
  try {
    const res = await fetchFn(url, {
      method: 'POST',
      headers: { 'Content-Type': 'text/plain;charset=UTF-8' },
      body: JSON.stringify({ publicKey, events }),
      credentials: 'omit',
    });
    if (res.ok) return 'ok';
    if (res.status === 429 || res.status >= 500) return 'retry';
    return 'drop';
  } catch {
    return 'retry';
  }
}

export function sendWithBeacon(url: string, publicKey: string, events: AnalyticsEvent[]): boolean {
  try {
    if (typeof navigator === 'undefined' || typeof navigator.sendBeacon !== 'function') return false;
    const blob = new Blob([JSON.stringify({ publicKey, events })], { type: 'text/plain;charset=UTF-8' });
    return navigator.sendBeacon(url, blob);
  } catch {
    return false;
  }
}

/** 1s, 2s, 4s, ... capped at 30s, plus up to 20% jitter so many clients don't retry in sync. */
export function backoffDelay(attempt: number, random: () => number = Math.random): number {
  const base = Math.min(30_000, 1000 * 2 ** (attempt - 1));
  return Math.round(base * (1 + random() * 0.2));
}
```

```ts
/** Calls onChange after pushState/replaceState/popstate (client-side navigation in SPAs like Next.js). */
export function watchLocation(onChange: () => void): () => void {
  const { pushState, replaceState } = history;
  // Run after the router has updated the page (and usually document.title).
  const notify = () => setTimeout(onChange, 0);

  history.pushState = function (...args: Parameters<History['pushState']>) {
    const result = pushState.apply(this, args);
    notify();
    return result;
  };
  history.replaceState = function (...args: Parameters<History['replaceState']>) {
    const result = replaceState.apply(this, args);
    notify();
    return result;
  };
  window.addEventListener('popstate', notify);

  return () => {
    history.pushState = pushState;
    history.replaceState = replaceState;
    window.removeEventListener('popstate', notify);
  };
}
```

```ts
const MAX_REQUEST_BYTES = 60_000; // stay under the server's 64 KB body limit

  private readonly identity = new Identity();
  private readonly queue = new PersistentQueue('acme_queue');
  private timer: ReturnType<typeof setInterval> | null = null;
  private cleanups: (() => void)[] = [];
  private inFlight: Promise<void> | null = null;
  private attempt = 0;
  private retryAt = 0;
  private lastPath: string | null = null;
  private lastHref: string | null = null;

  constructor(config: BrowserConfig) {

      flushInterval: 5000,
      maxBatchSize: 20,

      respectDoNotTrack: true,
      ...config,
      endpoint: config.endpoint.replace(/\/+$/, ''),
      fetch: config.fetch ?? ((input, init) => fetch(input, init)),

    this.timer = setInterval(() => void this.flush(), this.cfg.flushInterval);

    const onVisibility = () => { if (document.visibilityState === 'hidden') this.flushWithBeacon(); };
    const onPageHide = () => this.flushWithBeacon();
    const onOnline = () => { this.retryAt = 0; void this.flush(); };
    document.addEventListener('visibilitychange', onVisibility);
    window.addEventListener('pagehide', onPageHide);
    window.addEventListener('online', onOnline);
    this.cleanups.push(
      () => document.removeEventListener('visibilitychange', onVisibility),
      () => window.removeEventListener('pagehide', onPageHide),
      () => window.removeEventListener('online', onOnline),
    );

      this.cleanups.push(watchLocation(() => this.autoPageView()));
      this.autoPageView();

  private autoPageView(): void {
    if (location.pathname !== this.lastPath) this.page();
  }

  page(properties: Props = {}): void {
    const referrer = this.lastHref ?? (document.referrer || null);
    this.lastPath = location.pathname;
    this.lastHref = location.href;
    this.enqueue('page_view', 'page_view', properties, { title: document.title || null, referrer });
  }

  track(name: string, properties: Props = {}): void {
    this.enqueue('track', name, properties);
  }

  identify(userId: string, traits: Props = {}): void {
    this.identity.setUserId(userId);
    this.enqueue('identify', 'identify', traits);
  }

  reset(): void {
    this.identity.reset();
  }

  optOut(): void {
    this.identity.setOptOut(true);
    this.queue.clear();
  }

  optIn(): void {
    this.identity.setOptOut(false);
  }

  getAnonymousId(): string {
    return this.identity.getAnonymousId();
  }

  private enqueue(type: EventType, name: string, properties: Props, extra: Partial<AnalyticsEvent> = {}): void {
    if (this.disabled) {
      this.log('tracking disabled (opt-out or Do Not Track), dropped', name);
      return;
    }
    const event: AnalyticsEvent = {
      event_id: uuid(),
      type,
      name,
      anonymous_id: this.identity.getAnonymousId(),
      user_id: this.identity.getUserId(),
      session_id: this.identity.getSessionId(),
      timestamp: new Date().toISOString(),
      url: location.pathname,
      title: null,
      referrer: null,
      properties,
      ...extra,
    };
    this.queue.push(event);
    this.log('queued', event);
    if (this.queue.size >= this.cfg.maxBatchSize) void this.flush();
  }

  /** Oldest events first, at most 50 and ~60 KB per request. */
  private takeBatch(): AnalyticsEvent[] {
    const batch: AnalyticsEvent[] = [];
    let bytes = 0;
    for (const event of this.queue.peek(MAX_REQUEST_EVENTS)) {
      const size = JSON.stringify(event).length;
      if (batch.length > 0 && bytes + size > MAX_REQUEST_BYTES) break;
      batch.push(event);
      bytes += size;
    }
    return batch;
  }

  /** Sends one batch. Only one request is in flight at a time; respects retry backoff. */
  flush(): Promise<void> {
    if (this.inFlight) return this.inFlight;
    if (Date.now() < this.retryAt) return Promise.resolve();
    const batch = this.takeBatch();
    if (batch.length === 0) return Promise.resolve();
    this.inFlight = this.send(batch).finally(() => {
      this.inFlight = null;
    });
    return this.inFlight;
  }

  private async send(batch: AnalyticsEvent[]): Promise<void> {
    const result = await sendBatch(this.cfg.fetch, this.collectUrl, this.cfg.publicKey, batch);
    if (result === 'retry') {
      this.attempt++;
      const delay = backoffDelay(this.attempt);
      this.retryAt = Date.now() + delay;
      this.log(`send failed, retrying in ${delay} ms (attempt ${this.attempt})`);
      return;
    }
    if (result === 'drop') this.log(`server rejected batch, dropping ${batch.length} events`);
    this.attempt = 0;
    this.retryAt = 0;
    this.queue.remove(batch.map((e) => e.event_id));
    if (this.queue.size >= this.cfg.maxBatchSize) setTimeout(() => void this.flush(), 0);
  }

  /**
   * Page is being hidden/closed: hand the queue to the browser via sendBeacon.
   * Events stay queued; if they are sent again later the server de-duplicates by event_id.
   */
  private flushWithBeacon(): void {
    const batch = this.takeBatch();
    if (batch.length > 0) sendWithBeacon(this.collectUrl, this.cfg.publicKey, batch);
  }

  private log(...args: unknown[]): void {
    if (this.cfg.debug) console.log('[acme-analytics]', ...args);
  }
}
```

```ts
import { randomUUID } from 'node:crypto';
import type { AnalyticsEvent, IngestResponse } from '@acme/shared';

export interface ServerAnalyticsOptions {
  secretKey: string;
  endpoint: string;
  timeoutMs?: number;
  maxRetries?: number;
  retryDelayMs?: number;
  fetch?: typeof fetch;
  logger?: Pick<Console, 'warn'>;
}

export interface TrackIds {
  anonymousId?: string | null;
  userId?: string | null;
}

export interface TrackResult {
  ok: boolean;
  status?: number;
  error?: string;
}

const sleep = (ms: number) => new Promise((resolve) => setTimeout(resolve, ms));

/**
 * Sends trusted business events (e.g. order_created) from your server.
 * track() never throws: analytics problems must not break business logic.
 */
export class ServerAnalytics {
  private readonly url: string;
  private readonly timeoutMs: number;
  private readonly maxRetries: number;
  private readonly retryDelayMs: number;
  private readonly fetchFn: typeof fetch;
  private readonly logger: Pick<Console, 'warn'>;

  constructor(private readonly options: ServerAnalyticsOptions) {
    this.url = `${options.endpoint.replace(/\/+$/, '')}/api/server-events`;
    this.timeoutMs = options.timeoutMs ?? 2000;
    this.maxRetries = options.maxRetries ?? 2;
    this.retryDelayMs = options.retryDelayMs ?? 200;
    this.fetchFn = options.fetch ?? ((input, init) => fetch(input, init));
    this.logger = options.logger ?? console;
  }

  async track(name: string, properties: Record<string, unknown> = {}, ids: TrackIds = {}): Promise<TrackResult> {
    const event: AnalyticsEvent = {
      event_id: randomUUID(), // same id on every retry, so the server de-duplicates
      type: 'track',
      name,
      anonymous_id: ids.anonymousId ?? null,
      user_id: ids.userId ?? null,
      timestamp: new Date().toISOString(),
      properties,
    };
    const result = await this.send(event);
    if (!result.ok) this.logger.warn(`[acme-analytics] failed to send "${name}": ${result.error}`);
    return result;
  }

  private async send(event: AnalyticsEvent): Promise<TrackResult> {
    let last: TrackResult = { ok: false, error: 'not sent' };
    for (let attempt = 0; attempt <= this.maxRetries; attempt++) {
      if (attempt > 0) await sleep(this.retryDelayMs * 2 ** (attempt - 1));
      const controller = new AbortController();
      const timer = setTimeout(() => controller.abort(), this.timeoutMs);
      try {
        const res = await this.fetchFn(this.url, {
          method: 'POST',
          headers: { 'Content-Type': 'application/json', Authorization: `Bearer ${this.options.secretKey}` },
          body: JSON.stringify({ events: [event] }),
          signal: controller.signal,
        });
        if (res.ok) {
          const body = (await res.json().catch(() => null)) as IngestResponse | null;
          if (body?.rejected?.length) return { ok: false, status: res.status, error: body.rejected[0].error };
          return { ok: true, status: res.status };
        }
        last = { ok: false, status: res.status, error: `HTTP ${res.status}` };
        if (res.status < 500 && res.status !== 429) return last; // retrying cannot help
      } catch (err) {
        const e = err as Error;
        last = { ok: false, error: e?.name === 'AbortError' ? `timeout after ${this.timeoutMs} ms` : String(e?.message ?? err) };
      } finally {
        clearTimeout(timer);
      }
    }
    return last;
  }
}
```

```ts
export interface OrderLine {
  itemId: string;
  name: string;
  price: number;
  quantity: number;
}

export interface Order {
  id: string;
  lines: OrderLine[];
  total: number;
  customer: string | null;
  createdAt: string;
}

export class OrderError extends Error {}

/** Demo storage: in memory, cleared on restart. Kept on globalThis to survive hot reloads. */
const store: Map<string, Order> = ((globalThis as { __orders?: Map<string, Order> }).__orders ??= new Map());

/** Prices come from the menu, never from the client. */
export function priceOrder(items: { itemId: unknown; quantity: unknown }[]): { lines: OrderLine[]; total: number } {
  if (items.length === 0) throw new OrderError('Cart is empty');
  const lines = items.map(({ itemId, quantity }) => {
    const item = typeof itemId === 'string' ? findMenuItem(itemId) : undefined;
    if (!item) throw new OrderError(`Unknown menu item: ${String(itemId)}`);
    if (!Number.isInteger(quantity) || (quantity as number) < 1 || (quantity as number) > 20) {
      throw new OrderError(`Invalid quantity for ${item.name}`);
    }
    return { itemId: item.id, name: item.name, price: item.price, quantity: quantity as number };
  });
  const total = Math.round(lines.reduce((sum, l) => sum + l.price * l.quantity, 0) * 100) / 100;
  return { lines, total };
}

export const saveOrder = (order: Order) => void store.set(order.id, order);
export const getOrder = (id: string) => store.get(id);
```

```ts
import { randomUUID } from 'node:crypto';
import type { ServerAnalytics } from '@acme/server-sdk';
import { OrderError, priceOrder, saveOrder, type Order } from './orders';

type Analytics = Pick<ServerAnalytics, 'track'>;
type Defer = (task: () => Promise<unknown>) => void;

const optionalString = (v: unknown) => (typeof v === 'string' && v.length > 0 && v.length <= 200 ? v : null);

/**
 * Creates the order, then schedules the analytics event with `defer` (Next.js `after()`),
 * so a slow or down analytics server never delays or fails the customer's order.
 */
export function createOrder(body: unknown, analytics: Analytics, defer: Defer): { status: number; body: unknown } {
  const input = (body ?? {}) as { items?: unknown; customer?: unknown; anonymousId?: unknown; userId?: unknown };
  if (!Array.isArray(input.items)) return { status: 400, body: { error: 'items must be an array' } };

  let priced;
  try {
    priced = priceOrder(input.items);
  } catch (err) {
    if (err instanceof OrderError) return { status: 400, body: { error: err.message } };
    throw err;
  }

  const order: Order = {
    id: `ord_${randomUUID().slice(0, 8)}`,
    lines: priced.lines,
    total: priced.total,
    customer: optionalString(input.customer),
    createdAt: new Date().toISOString(),
  };
  saveOrder(order);

  const ids = { anonymousId: optionalString(input.anonymousId), userId: optionalString(input.userId) };
  defer(() =>
    analytics.track(
      'order_created',
      { orderId: order.id, total: order.total, itemCount: order.lines.reduce((n, l) => n + l.quantity, 0), items: order.lines.map((l) => l.itemId) },
      ids,
    ),
  );

  return { status: 201, body: { orderId: order.id, total: order.total } };
}
```

```ts
'use client';
import { useCallback, useEffect, useState } from 'react';

export interface CartLine {
  itemId: string;
  quantity: number;
}

const KEY = 'restaurant_cart';
const EVENT = 'cart-change';

function read(): CartLine[] {
  try {
    const parsed = JSON.parse(localStorage.getItem(KEY) ?? '[]');
    return Array.isArray(parsed) ? parsed : [];
  } catch {
    return [];
  }
}

function write(lines: CartLine[]) {
  try {
    localStorage.setItem(KEY, JSON.stringify(lines));
  } catch {
    /* ignore */
  }
  window.dispatchEvent(new Event(EVENT));
}

export function useCart() {
  const [lines, setLines] = useState<CartLine[]>([]);

  useEffect(() => {
    const sync = () => setLines(read());
    sync();
    window.addEventListener(EVENT, sync);
    window.addEventListener('storage', sync);
    return () => {
      window.removeEventListener(EVENT, sync);
      window.removeEventListener('storage', sync);
    };
  }, []);

  const add = useCallback((itemId: string) => {
    const current = read();
    const line = current.find((l) => l.itemId === itemId);
    if (line) line.quantity = Math.min(20, line.quantity + 1);
    else current.push({ itemId, quantity: 1 });
    write(current);
  }, []);

  const remove = useCallback((itemId: string) => write(read().filter((l) => l.itemId !== itemId)), []);
  const clear = useCallback(() => write([]), []);
  const count = lines.reduce((n, l) => n + l.quantity, 0);

  return { lines, count, add, remove, clear };
}
```

```ts
    const sync = () => setUser(read());

    window.addEventListener(EVENT, sync);

    reset();
    window.dispatchEvent(new Event(EVENT));
  }, []);
```

```tsx
'use client';
import { useEffect, useRef, useState } from 'react';
import { useRouter } from 'next/navigation';
import { getAnonymousId, track } from '@acme/browser-sdk';
import { findMenuItem } from '../../lib/menu';
import { useCart } from '../../lib/cart';
import { useUser } from '../../lib/user';

export default function CheckoutPage() {
  const router = useRouter();
  const { lines, clear } = useCart();
  const { user } = useUser();
  const [name, setName] = useState('');
  const [error, setError] = useState<string | null>(null);
  const [submitting, setSubmitting] = useState(false);
  const tracked = useRef(false);

  const total = lines.reduce((sum, l) => sum + (findMenuItem(l.itemId)?.price ?? 0) * l.quantity, 0);

  useEffect(() => {
    if (user && !name) setName(user.name);
  }, [user, name]);

  useEffect(() => {
    if (lines.length > 0 && !tracked.current) {
      tracked.current = true;
      track('checkout_started', { itemCount: lines.reduce((n, l) => n + l.quantity, 0), total });
    }
  }, [lines, total]);

  async function placeOrder(e: React.FormEvent) {
    e.preventDefault();
    setSubmitting(true);
    setError(null);
    try {
      const res = await fetch('/api/orders', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        // anonymousId links this server-side order to the visitor's browser journey.
        body: JSON.stringify({ items: lines, customer: name, anonymousId: getAnonymousId(), userId: user?.id ?? null }),
      });
      const body = await res.json();
      if (!res.ok) throw new Error(body.error ?? 'Could not place order');
      clear();
      router.push(`/order/${body.orderId}`);
    } catch (err) {
      setError((err as Error).message);
      setSubmitting(false);
    }
  }

  if (lines.length === 0) return <main><h1>Checkout</h1><p className="note">Your cart is empty.</p></main>;

  return (
    <main>
      <h1>Checkout</h1>
      <form onSubmit={placeOrder} style={{ display: 'flex', flexDirection: 'column', gap: 12, maxWidth: 360 }}>
        <label>Name for the order<br /><input value={name} onChange={(e) => setName(e.target.value)} required /></label>
        <div className="total" style={{ textAlign: 'left' }}>Total: ${total.toFixed(2)}</div>
        {error && <div className="error">{error}</div>}
        <button className="btn" type="submit" disabled={submitting}>{submitting ? 'Placing order…' : 'Place order'}</button>
      </form>
    </main>
  );
}
```
